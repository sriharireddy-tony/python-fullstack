JavaScript semantics · Problem 07

Closures and the Counter Factory

A closure captures a variable, not a value. One sentence, and every confusing closure question — the loop that prints 3 three times, the counter that shares state, the leak that keeps a megabyte alive — follows from it.

Step 1

The problem — what a closure is

The definition

A closure is a function together with the variables it can still reach from where it was defined, even after that outer function has returned.

The part everyone under-weights: it captures the variable itself, a live binding — not a snapshot of what the variable held at the time.

variable, not value — the proof
function makeReader() {
  let x = 1;
  const read = () => x;
  x = 2;                     // changed AFTER the arrow was created
  return { read, set: (v) => { x = v; } };
}

const r = makeReader();
console.log('read()      ->', r.read());   // 2, not 1
r.set(99);
console.log('after set   ->', r.read());   // 99

// two functions closing over the SAME variable see each other's writes
function makePair() {
  let shared = 0;
  return {
    inc: () => ++shared,
    get: () => shared,
  };
}
const pair = makePair();
pair.inc(); pair.inc(); pair.inc();
console.log('');
console.log('shared between two closures:', pair.get());   // 3

// but two CALLS to the factory get independent variables
const a = makePair(), b = makePair();
a.inc(); a.inc();
b.inc();
console.log('a:', a.get(), '  b:', b.get());   // 2  1
console.log('');
console.log('One factory call = one set of variables = one closure');
console.log('scope. That is the whole mechanism.');

// Write a counter with a count nobody outside can touch.

Think first — the alternatives, and what they give up

four ways to hold a count
// 1. a module-level variable -- shared by everyone, no privacy
let globalCount = 0;
const globalIncrement = () => ++globalCount;

// 2. a plain object property -- private in name only
function makeConventional() {
  return { _count: 0, increment() { return ++this._count; } };
}

// 3. a closure -- genuinely private
function makeClosure() {
  let count = 0;
  return { increment: () => ++count, value: () => count };
}

// 4. a class with a #private field -- private, and inspectable as a class
class ClassCounter {
  #count = 0;
  increment() { return ++this.#count; }
  get value() { return this.#count; }
}

globalIncrement();
const conv = makeConventional();
const clos = makeClosure();
const cls = new ClassCounter();
conv.increment(); clos.increment(); cls.increment();

console.log('1. global    :', globalCount,
            '  tampering: globalCount = 500 works, obviously');
conv._count = 500;
console.log('2. underscore:', conv._count, '  tampering: also works');
console.log('3. closure   :', clos.value(), '  tampering: nothing to tamper with');
try { cls.count = 500; } catch (e) { /* ignored */ }
console.log('4. #private  :', cls.value, '  tampering:');
try {
  eval('cls.#count = 500');
} catch (e) {
  console.log('     cls.#count = 500 ->', e.constructor.name,
              '-- a SYNTAX error, not even runtime');

Step 3

Why it matters — the memory cost

A closure keeps its whole scope alive. Every instance of a closure-based object holds its own copy of every method; a class shares one copy on the prototype.

Memory demo — closures versus a prototype
function makeClosureCounter() {
  let count = 0;
  return {
    increment() { return ++count; },
    decrement() { return --count; },
    value() { return count; },
    reset() { count = 0; },
  };
}

class ClassCounter {
  constructor() { this._count = 0; }
  increment() { return ++this._count; }
  decrement() { return --this._count; }
  value() { return this._count; }
  reset() { this._count = 0; }
}

const N = 200000;

const before1 = process.memoryUsage ? process.memoryUsage().heapUsed : 0;

Step 4

The better idea — var, let, and the loop

The single most-asked closure question. The answer is one fact about scope.

  for (var i = 0; i < 3; i++) {
      setTimeout(() => console.log(i), 0);
  }

  `var` is FUNCTION-scoped. There is exactly ONE `i`, shared by
  the loop and by all three callbacks:

      +---------------- function scope ----------------+
      |  var i  --------------------------> 3          |
      +------------------------------------------------+
           ^            ^            ^
           |            |            |
        cb#1         cb#2         cb#3     all point at the SAME box

  By the time the callbacks run, the loop has finished and the
  one shared `i` holds 3. Output: 3 3 3.


  for (let i = 0; i < 3; i++) { ... }

  `let` is BLOCK-scoped, and the specification gives a for-loop a
  FRESH binding per iteration, copying the value forward:

      +-- iteration 0 --+  +-- iteration 1 --+  +-- iteration 2 --+
      |  let i = 0      |  |  let i = 1      |  |  let i = 2      |
      +-----------------+  +-----------------+  +-----------------+
             ^                     ^                     ^
           cb#1                  cb#2                  cb#3

  Three separate boxes. Output: 0 1 2.
all of it, side by side
const out = { var: [], let: [], iife: [], bind: [], forEach: [] };

for (var i = 0; i < 3; i++) setTimeout(() => out.var.push(i), 0);

for (let j = 0; j < 3; j++) setTimeout(() => out.let.push(j), 0);

// the pre-ES6 fix: an IIFE makes a new function scope per iteration
for (var k = 0; k < 3; k++) {
  (function (captured) {
    setTimeout(() => out.iife.push(captured), 0);
  })(k);
}

// bind also works -- the argument is copied at bind time
for (var m = 0; m < 3; m++) {
  setTimeout(function (captured) { out.bind.push(captured); }.bind(null, m), 0);
}

// forEach gives each callback its own parameter, so var is harmless
[0, 1, 2].forEach(function (n) { setTimeout(() => out.forEach.push(n), 0); });

await new Promise((r) => setTimeout(r, 40));

console.log('var     :', out.var);
console.log('let     :', out.let);
console.log('IIFE    :', out.iife);
console.log('bind    :', out.bind);
console.log('forEach :', out.forEach);
console.log('');
console.log('Everything except the first line works, and they all work');
console.log('for the same reason: each callback ends up with its own');
console.log('binding instead of sharing one.');

// and the detail that surprises people: `var` leaks out of the block
console.log('');
console.log('after the loops, i is', i, '-- var escaped the loop');
try { console.log('j is', j); }
catch (e) { console.log('j is', e.constructor.name + ' -- let did not'); }
The one case where let does not save you

The fresh-binding-per-iteration rule is specific to for (let i = ...). Declare the variable outside the loop and let behaves exactly like var — one binding, shared. Shown in Step 9.

Step 5

Dry run — watch the scope chain

  function outer() {
      let a = 1;
      function middle() {
          let b = 2;
          function inner() {
              let c = 3;
              return a + b + c;
          }
          return inner;
      }
      return middle();
  }
  const fn = outer();
  fn();

  When inner() runs, `a` and `b` are looked up along the SCOPE
  CHAIN -- the chain of environments that existed where inner
  was WRITTEN, not where it is CALLED:

      inner's own scope    { c: 3 }
              |  not found? go up
      middle's scope       { b: 2, inner }
              |  not found? go up
      outer's scope        { a: 1, middle }
              |  not found? go up
      module / global scope
              |
      not found -> ReferenceError

  outer() and middle() have both RETURNED by the time fn() runs.
  Their scopes survive anyway, because inner still references
  them. That is the closure.
the chain, narrating itself
function outer() {
  let a = 1;
  console.log('outer running. a =', a);

  function middle() {
    let b = 2;
    console.log('  middle running. b =', b, ' (a is still reachable:', a + ')');

    function inner() {
      let c = 3;
      console.log('    inner running. c =', c);
      console.log('    reaching up for b:', b);
      console.log('    reaching up for a:', a);
      return a + b + c;
    }

    return inner;
  }

  const fn = middle();
  console.log('outer returning. middle() has already finished.');
  return fn;
}

console.log('');
console.log('And the scopes are LIVE, not frozen -- a setter proves it:');

Step 6

Version 1 — the patterns built on closures

1. once — run a function at most one time
function once(fn) {
  let called = false;
  let result;

  return function (...args) {
    if (called) return result;          // every later call replays the result
    called = true;
    result = fn.apply(this, args);
    fn = null;                          // release the reference, help the GC
    return result;
  };
}

let initCount = 0;
const initialise = once((name) => {
  initCount++;
  return 'initialised ' + name;
});

console.log(initialise('app'));
console.log(initialise('again'));       // same result, fn not re-run
console.log(initialise('and again'));
console.log('the wrapped function ran', initCount, 'time(s)');

console.log('');
console.log('Note `called` rather than `result === undefined`: a');
console.log('function that legitimately returns undefined must still');
console.log('only run once.');
let voidRuns = 0;
const undef = once(() => { voidRuns++; return undefined; });
console.log('  three calls returned:', undef(), undef(), undef());
console.log('  and the function ran', voidRuns, 'time');
console.log('  A `result === undefined` check would have re-run it');
console.log('  every single time.');

// a limiter: allow n calls, then refuse
function limitCalls(fn, max) {
  let used = 0;
  return function (...args) {
    if (used >= max) throw new Error('call limit of ' + max + ' reached');
    used++;
    return fn.apply(this, args);
  };
}

const send = limitCalls((msg) => 'sent: ' + msg, 2);

Step 8

Complexity, in sentences

Closures cost nothing in time — a variable lookup up the scope chain is a compile-time resolved slot in every modern engine, not a search. The cost is memory, and lifetime.

WhatCost
Creating a closureone function object plus a pointer to its scope
Reading a captured variableO(1) — a slot, not a lookup
n closure-based objects with m methodsO(n · m) function objects
n class instances with m methodsO(n) objects, m functions total
Lifetime of a captured variableas long as any closure over it lives

That last row is the one that causes bugs. A single surviving closure keeps its entire scope alive — not only the variables it uses, in principle, though modern engines do prune variables the closure demonstrably never mentions. Anything reachable through a name the closure does mention is retained for as long as the closure is.

Say this in the interview

“A closure is a function plus the scope it was defined in, and it captures the variable rather than its value — so if the variable changes later, the closure sees the new value. That is why var in a loop gives you three functions sharing one binding that ends at 3, while for (let i ...) creates a fresh binding per iteration. The practical cost is lifetime: the captured scope lives as long as the closure does, so an event listener or a pending timer can keep a large object alive indefinitely.”

3. Forgetting to remove a listener

Broken — the handler outlives the component
// a tiny event emitter, so this runs anywhere
function makeEmitter() {
  const listeners = [];
  return {
    on: (fn) => { listeners.push(fn); return () => {
      const i = listeners.indexOf(fn);
      if (i >= 0) listeners.splice(i, 1);
    }; },
    emit: (v) => listeners.forEach((fn) => fn(v)),
    count: () => listeners.length,
  };
}

const bus = makeEmitter();

function mountLeaky(id) {
  const state = { id, buffer: new Array(1000).fill(id) };
  bus.on(() => { state.buffer[0] = 'touched'; });
  // no way to remove it -- the reference was never kept
}

function mountClean(id) {
  const state = { id, buffer: new Array(1000).fill(id) };
  const off = bus.on(() => { state.buffer[0] = 'touched'; });
  return off;                       // the caller can unsubscribe
}

for (let i = 0; i < 5; i++) mountLeaky(i);
console.log('after 5 leaky mounts, listeners:', bus.count());

const offs = [];
for (let i = 0; i < 5; i++) offs.push(mountClean(i));

4. Sharing state you meant to keep separate

Broken — one counter pretending to be many
// the factory is fine
function makeCounter() {
  let count = 0;
  return () => ++count;
}

// but calling it ONCE and reusing the result is not
const shared = makeCounter();
const a = shared;
const b = shared;
console.log('a():', a(), a());
console.log('b():', b(), '  <- continued from a, not a fresh counter');

// the fix
const c = makeCounter();
const d = makeCounter();
console.log('');
console.log('c():', c(), c());
console.log('d():', d(), '  <- independent');

// the same bug in a module
console.log('');
const CounterModule = (function () {
  let count = 0;                       // ONE count for the whole module
  return { increment: () => ++count };
})();
console.log('module counter, "two" users:',
            CounterModule.increment(), CounterModule.increment());
console.log('An IIFE module has exactly one instance, by construction.');
console.log('If you need many, export the FACTORY, not the result.');

5. Expecting a closure over an object to be a snapshot

Broken — the object mutated underneath
function makeReport(user) {
  return () => 'report for ' + user.name;
}

const user = { name: 'Ada' };
const report = makeReport(user);
console.log('before:', report());

user.name = 'Grace';
console.log('after :', report(), '  <- the object was mutated');

console.log('');
console.log('The PARAMETER `user` was copied -- but what was copied is');
console.log('a reference to the same object. Reassigning the outer');
console.log('variable does NOT show through:');

let outer = { name: 'Ada' };
const report2 = makeReport(outer);
outer = { name: 'Replaced' };            // a new object in the outer variable
console.log('  after reassigning outer:', report2());

console.log('');
console.log('If you want a snapshot, take one:');
function makeSnapshotReport(user) {
  const name = user.name;                // copy the primitive out
  return () => 'report for ' + name;
}
const u2 = { name: 'Ada' };
const snap = makeSnapshotReport(u2);
u2.name = 'Grace';
console.log(' ', snap());

console.log('');
console.log('Two different questions, always worth separating:');
console.log('  is the VARIABLE shared?  (var vs let, same scope or not)');
console.log('  is the VALUE shared?     (object vs primitive)');

What to carry forward

Captures a variable, not a value. Say it out loud until it is automatic. Then two corollaries: variables in the same scope are shared, so the question is always “how many scopes did this code create?”; and a captured scope lives exactly as long as the closure does, which is where the leaks come from.
