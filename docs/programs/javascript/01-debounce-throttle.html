JavaScript semantics · Problem 01

Debounce and Throttle

Two functions that both “slow down” a callback, and are constantly confused for each other. They answer different questions: debounce asks have they stopped yet?, throttle asks has enough time passed? Getting the difference right is the whole interview.

Step 1

The problem

A user types in a search box. Each keystroke fires a handler that calls your API. Thirty keystrokes, thirty requests, twenty-nine of them already stale by the time they land.

Two different fixes, for two different situations:

DebounceThrottle
Rulerun once, d ms after the last callrun at most once every d ms
During a burstnothing runsruns on a steady beat
Question it answershave they finished?has enough time passed?
Use forsearch-as-you-type, autosave, resize-end, form validationscroll position, mousemove, progress bars, analytics pings
Dangera never-ending burst means it never runsyou may miss the final value
The one-line version to remember

Debounce waits for quiet. Throttle keeps a rhythm.

Or in code terms: debounce resets its timer on every call; throttle ignores calls until its timer expires.

the difference, visibly
function debounce(fn, delay) {
  let timer = null;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

function throttle(fn, interval) {
  let last = 0;
  return function (...args) {
    const now = Date.now();
    if (now - last >= interval) {
      last = now;
      fn.apply(this, args);
    }
  };
}

}, 100);

  console.log('  THROTTLE fired at ~' + (Date.now() - start) + ' ms with "' + ch + '"');

Step 4

The better idea — a closure over the timer

Both functions have the same shape: a factory that returns a wrapper, with private state trapped between them.

  function debounce(fn, delay) {

      let timer = null;          <-- private state, one per call
                                     to debounce()
      return function (...args) {
          clearTimeout(timer);   <-- cancel the pending run
          timer = setTimeout(    <-- schedule a new one
              () => fn(...args),
              delay
          );
      };
  }

  The returned function OUTLIVES debounce(), but it still sees
  `timer`. That is a closure: the inner function keeps the
  variable alive, and each call to debounce() creates a fresh one.
The two mechanisms, side by side

Debounce keeps a timer id and clears it. Every call cancels the pending run and starts over, so only quiet lets one through.

Throttle keeps a timestamp and compares it. Calls arriving inside the window are simply dropped — nothing is cancelled, because nothing was scheduled.

Because the state is per-wrapper, two independently limited functions never interfere:

each wrapper gets its own state
function debounce(fn, delay) {

  console.log('  debounce() called -- new private `timer` created');

    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

Version 1 — the plain versions, written out

Version 1a — debounce, no arrow functions or rest args
function debounce(fn, delay) {
  var timer = null;

  return function () {
    var context = this;                  // remember the receiver
    var args = arguments;                // and the arguments

    clearTimeout(timer);
    timer = setTimeout(function () {
      fn.apply(context, args);           // restore both
    }, delay);
  };
}

// Why `context` and `args` are captured into variables:
// a plain `function () {}` inside setTimeout gets its OWN `this`
// and its OWN `arguments`. Arrow functions inherit both, which is
// why the modern version can skip this step.

const log = [];
const d = debounce(function (v) { log.push(v + ' / this.name=' + this.name); }, 50);

const widget = { name: 'searchBox', run: d };
widget.run('typed');

await sleep(120);

console.log(log);
// ['typed / this.name=searchBox']  -- the receiver survived the timer
Version 1b — throttle, two implementations
// (i) timestamp style -- fires IMMEDIATELY on the first call
function throttleByTime(fn, interval) {
  var last = 0;
  return function () {
    var now = Date.now();
    if (now - last >= interval) {
      last = now;
      fn.apply(this, arguments);
    }
  };
}

// (ii) timer style -- fires on the first call, then blocks until
//      the timer clears. Equivalent in effect, different in feel.
function throttleByTimer(fn, interval) {
  var waiting = false;
  return function () {
    if (waiting) return;
    waiting = true;
    fn.apply(this, arguments);
    setTimeout(function () { waiting = false; }, interval);
  };
}

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

const ta = throttleByTime((v) => a.push(v), 100);
const tb = throttleByTimer((v) => b.push(v), 100);

for (let i = 1; i <= 12; i++) {
  ta(i); tb(i);
  await sleep(30);
}
await sleep(150);

console.log('  by timestamp:', a);
console.log('  by timer    :', b);

Version 2 — the versions worth shipping

The eight-line versions are the right first answer. Then the interviewer asks for the options, and this is what they want.

Debounce with leading, trailing, cancel and flush
function debounce(fn, delay, options = {}) {
  const leading  = options.leading  === true;    // fire on the first call
  const trailing = options.trailing !== false;   // fire after the quiet period

  let timer = null;
  let lastArgs = null;
  let lastThis = null;
  let calledDuringWait = false;

  function invoke() {
    const args = lastArgs, ctx = lastThis;
    lastArgs = lastThis = null;
    return fn.apply(ctx, args);
  }

  function wrapped(...args) {
    lastArgs = args;
    lastThis = this;

    const isFirst = timer === null;

    if (isFirst && leading) {
      invoke();
    } else {
      calledDuringWait = true;
    }

    clearTimeout(timer);
    timer = setTimeout(() => {
      timer = null;
      if (trailing && calledDuringWait) invoke();
      calledDuringWait = false;
    }, delay);
  }

  wrapped.cancel = function () {
    clearTimeout(timer);
    timer = null;
    lastArgs = lastThis = null;
    calledDuringWait = false;
  };

  wrapped.flush = function () {
    if (timer !== null && lastArgs !== null) {
      clearTimeout(timer);
      timer = null;
      calledDuringWait = false;
      return invoke();
    }
  };

  wrapped.pending = function () { return timer !== null; };

  return wrapped;
}

Throttle that does not drop the last event
// The classic throttle bug: the user stops scrolling mid-window, so
// the final position never reaches your handler. Fix it by keeping
// the last arguments and firing them when the window closes.
function throttle(fn, interval, options = {}) {
  const leading  = options.leading  !== false;   // default true
  const trailing = options.trailing !== false;   // default true

  let last = 0;
  let timer = null;
  let lastArgs = null;
  let lastThis = null;

  function wrapped(...args) {
    const now = Date.now();
    if (last === 0 && !leading) last = now;      // pretend we just fired

    const remaining = interval - (now - last);

    if (remaining <= 0) {
      if (timer) { clearTimeout(timer); timer = null; }
      last = now;
      fn.apply(this, args);
    } else if (trailing && timer === null) {
      lastArgs = args;
      lastThis = this;
      timer = setTimeout(() => {
        last = options.leading === false ? 0 : Date.now();
        timer = null;
        fn.apply(lastThis, lastArgs);
        lastArgs = lastThis = null;
      }, remaining);
    } else if (trailing) {
      lastArgs = args;                            // keep the freshest values
      lastThis = this;
    }
  }

  wrapped.cancel = function () {
    clearTimeout(timer);
    timer = null;
    last = 0;
    lastArgs = lastThis = null;
  };

  return wrapped;
}

function simpleThrottle(fn, interval) {
  let last = 0;
  return function (...args) {
    const now = Date.now();
    if (now - last >= interval) { last = now; fn.apply(this, args); }
  };
}

// For anything that ends in a paint -- moving an element, reading
// layout, drawing -- requestAnimationFrame beats a timer. It runs
// at most once per frame and is aligned WITH the paint, so you can
// never queue two updates for one frame.
function rafThrottle(fn) {
  let queued = false;
  let lastArgs = null;

  return function (...args) {
    lastArgs = args;
    if (queued) return;
    queued = true;
    requestAnimationFrame(() => {
      queued = false;
      fn.apply(this, lastArgs);
    });
  };
}

// Node has no requestAnimationFrame, so this demo fakes one at
// 60 fps to show the shape. In a browser, delete the shim.
const raf = typeof requestAnimationFrame === 'function'
  ? requestAnimationFrame
  : (cb) => setTimeout(() => cb(Date.now()), 16);

let queued = false, lastPos = null, painted = 0;
function onScroll(p) {
  lastPos = p;
  if (queued) return;
  queued = true;
  raf(() => { queued = false; painted++; });
}

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

let events = 0;

await sleep(60);

Step 8

Complexity, in sentences

Neither one has interesting time complexity — the wrapper is O(1) per call. What matters is how many times the wrapped function runs, and that is what you should quantify.

Over n events in T ms, window dRuns
Rawn
Throttleabout T / d
Debounceone per quiet gap — often 1

Space is O(1) per wrapper: a timer id, a timestamp, and the last arguments. Note the last of those — a debounced function holds a reference to its most recent arguments and receiver until it fires. Debounce a handler that receives a large object and you are keeping that object alive for the length of the delay.

Say this in the interview

“Both are factories that return a wrapper closing over private state. Debounce keeps a timer id, clears it on every call and reschedules, so it only runs after the calls stop — right for search-as-you-type or autosave. Throttle keeps a timestamp and drops calls inside the window, so it runs on a steady beat — right for scroll or mousemove. I would use apply with the saved this and arguments so the wrapper is transparent to the caller, and I would add a trailing call to the throttle, because otherwise the final event of a burst is silently dropped.”

Broken — two ways

function debounceLostThis(fn, delay) {
  let timer = null;

    clearTimeout(timer);

5. Forgetting to cancel on teardown

Broken — a callback that outlives its component
function debounce(fn, delay) {
  let timer = null;
  const wrapped = function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
  wrapped.cancel = () => clearTimeout(timer);
  return wrapped;
}

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

// pretend this is a component that gets unmounted
function makeComponent(name, cancelOnUnmount) {
  let alive = true;
  const save = debounce((text) => {
    if (!alive) {
      console.log('  [' + name + '] setState on an unmounted component!');
      return;
    }
    console.log('  [' + name + '] saved:', text);
  }, 80);

  return {
    type: (text) => save(text),
    unmount: () => {
      alive = false;
      if (cancelOnUnmount) save.cancel();
      console.log('  [' + name + '] unmounted' +
                  (cancelOnUnmount ? ' (cancelled the pending save)' : ''));
    },
  };
}

const leaky = makeComponent('leaky', false);
leaky.type('half a sentence');
leaky.unmount();
await sleep(150);

const clean = makeComponent('clean', true);
clean.type('half a sentence');
clean.unmount();
await sleep(150);

console.log('');
console.log('The pending timer holds the callback, which holds the');
console.log('closure, which holds whatever the component captured.');
console.log('Always expose cancel(), and always call it on teardown --');
console.log('in React, the cleanup function of the effect.');
