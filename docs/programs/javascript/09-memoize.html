JavaScript semantics · Problem 09

Memoize a Function

Six lines get you a working memoize. Then the interviewer asks what happens with two arguments, or an object argument, or a function that returns a promise, or a cache that grows forever — and every one of those is really a question about the key.

Step 1

The problem

Wrap a function so that calling it twice with the same arguments does the work only once.

The precondition nobody states

Memoization is only correct for a pure function: same input, same output, no side effects. Memoize something that reads a clock, a database, or a mutable object and you have not made it faster — you have made it wrong.

Version 0 — the six-line version
function memoize(fn) {
  const cache = new Map();

  return function (...args) {
    const key = JSON.stringify(args);
    if (cache.has(key)) return cache.get(key);
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

Step 3

Why the key strategy decides everything

Timing demo — four key strategies
function makeMemo(keyFn) {
  return function (fn) {
    const cache = new Map();
    return (...args) => {
      const k = keyFn(args);
      if (cache.has(k)) return cache.get(k);
      const r = fn(...args);
      cache.set(k, r);
      return r;
    };
  };
}

const work = (a, b) => { let s = 0; for (let i = 0; i < 200; i++) s += a * b; return s; };
const N = 200000;

const strategies = {
  'no memo         ': null,
  'JSON.stringify  ': makeMemo((args) => JSON.stringify(args)),
  'join with |     ': makeMemo((args) => args.join('|')),
  'single arg only ': makeMemo((args) => args[0]),
  'nested Map      ': null,   // handled separately below
};

for (const [label, wrap] of Object.entries(strategies)) {
  if (label.startsWith('nested')) continue;
  const fn = wrap ? wrap(work) : work;
  const t = performance.now();
  for (let i = 0; i < N; i++) fn(i % 50, 7);
  console.log(label + ': ' + (performance.now() - t).toFixed(1) + ' ms');
}

// a nested Map: one level per argument, no string building at all
function memoNested(fn) {
  const root = new Map();
  const MISS = Symbol('miss');
  return (...args) => {
    let node = root;
    for (const a of args) {
      if (!node.has(a)) node.set(a, new Map());
      node = node.get(a);
    }
    if (node.has(MISS) === false && node.has('value')) return node.get('value');
    const r = fn(...args);
    node.set('value', r);
    return r;
  };
}
const nested = memoNested(work);
let t = performance.now();
for (let i = 0; i < N; i++) nested(i % 50, 7);
console.log('nested Map      : ' + (performance.now() - t).toFixed(1) + ' ms');

console.log('');
console.log('The wrapper overhead is charged on EVERY call, hit or');
console.log('miss. JSON.stringify is the most general and the most');
console.log('expensive; a nested Map avoids strings entirely and keys');
console.log('on identity, which is both faster and more correct.');
console.log('');
console.log('Rule: memoize when the function is expensive relative to');
console.log('the key. Never memoize an addition.');

Why a Map and not a plain object

A plain object stringifies every key, so 1 and '1' collide. It also inherits from Object.prototype, so cache['toString'] is a false hit before anything is stored. A Map has neither problem, keeps insertion order, and has an O(1) size.

the plain-object cache, failing twice
function memoWithObject(fn) {
  const cache = {};
  return (x) => {
    if (cache[x] !== undefined) return cache[x];
    return (cache[x] = fn(x));
  };
}

function memoWithMap(fn) {
  const cache = new Map();
  return (x) => {
    if (cache.has(x)) return cache.get(x);
    const r = fn(x);
    cache.set(x, r);

}

the algorithm, narrating itself
function memoizeLoud(fn, resolver) {
  const cache = new Map();

  const wrapped = function (...args) {
    const key = resolver ? resolver(...args) : JSON.stringify(args);

    if (cache.has(key)) {
      console.log('  key ' + JSON.stringify(key) + '  HIT  -> ' + cache.get(key));
      return cache.get(key);
    }

    console.log('  key ' + JSON.stringify(key) + '  MISS -- computing');
    const result = fn.apply(this, args);
    cache.set(key, result);
    console.log('    stored. cache size is now ' + cache.size);
    return result;
  };

  wrapped.cache = cache;
  return wrapped;
}

// A Map keeps its keys alive forever. A WeakMap does not: when the
// object is garbage, so is its cache entry.
function memoizeWeak(fn) {
  const cache = new WeakMap();
  return (obj) => {
    if (cache.has(obj)) return cache.get(obj);
    const r = fn(obj);
    cache.set(obj, r);
    return r;
  };
}

3. a bounded cache — LRU
// An unbounded memo on user input is a memory leak with a friendly
// name. Cap it, and evict the least recently used entry.
function memoizeLRU(fn, maxSize = 3) {
  const cache = new Map();       // Map preserves INSERTION order

  return function (...args) {
    const key = args.join('');

    if (cache.has(key)) {
      const value = cache.get(key);
      cache.delete(key);         // delete + set moves it to the end,
      cache.set(key, value);     // making it the most recently used
      return value;
    }

    const result = fn.apply(this, args);
    cache.set(key, result);

    if (cache.size > maxSize) {
      // the first key in iteration order is the oldest
      const oldest = cache.keys().next().value;
      cache.delete(oldest);
    }

    return result;
  };
}

Version 2 — the one to write in the interview

Version 2 — resolver, cache access, and a size cap
function memoize(fn, options = {}) {
  const resolver = options.resolver;
  const maxSize = options.maxSize ?? Infinity;
  const cache = new Map();

  function defaultKey(args) {
    // fast path for the overwhelmingly common cases
    if (args.length === 0) return 'empty';
    if (args.length === 1) {
      const a = args[0];
      if (a === null || typeof a !== 'object') return typeof a + ':' + String(a);
    }
    return JSON.stringify(args);
  }

  function wrapped(...args) {
    const key = resolver ? resolver.apply(this, args) : defaultKey(args);

    if (cache.has(key)) {
      const value = cache.get(key);
      if (maxSize !== Infinity) { cache.delete(key); cache.set(key, value); }
      return value;
    }

    const result = fn.apply(this, args);
    cache.set(key, result);

    if (cache.size > maxSize) cache.delete(cache.keys().next().value);
    return result;
  }

  wrapped.cache = cache;
  wrapped.clear = () => cache.clear();
  wrapped.delete = (...args) =>
    cache.delete(resolver ? resolver(...args) : defaultKey(args));

  return wrapped;
}

memoizing an async function — cache the PROMISE
// The subtle part: cache the promise, not the resolved value.
// Caching the value means N concurrent callers all start N
// requests before the first one finishes.
let requests = 0;
const fetchUser = (id) => {
  requests++;
  return new Promise((r) => setTimeout(() => r({ id, name: 'user' + id }), 20));
};

// WRONG: awaits inside, so the cache is only written on completion
function memoAsyncWrong(fn) {
  const cache = new Map();
  return async (id) => {
    if (cache.has(id)) return cache.get(id);
    const value = await fn(id);          // a gap, during which nothing is cached
    cache.set(id, value);
    return value;
  };
}

// RIGHT: store the promise immediately
function memoAsync(fn) {
  const cache = new Map();
  return (id) => {
    if (cache.has(id)) return cache.get(id);
    const promise = fn(id).catch((err) => {
      cache.delete(id);                  // do not cache a failure forever
      throw err;
    });
    cache.set(id, promise);              // stored BEFORE anyone awaits
    return promise;
  };
}

a stable key for object arguments
// JSON.stringify depends on key insertion order. If you need
// structural equality, sort the keys as you serialise.
function stableStringify(value) {
  const seen = new WeakSet();

  return (function walk(v) {
    if (v === null) return 'null';
    if (typeof v === 'number') return Object.is(v, -0) ? '-0' : String(v);
    if (typeof v === 'bigint') return v + 'n';
    if (typeof v === 'string') return JSON.stringify(v);
    if (typeof v === 'boolean' || typeof v === 'undefined') return String(v);
    if (typeof v === 'function') throw new TypeError('cannot key on a function');

    if (seen.has(v)) throw new TypeError('cannot key on a cyclic structure');
    seen.add(v);

    let out;
    if (Array.isArray(v)) {
      out = '[' + v.map(walk).join(',') + ']';
    } else if (v instanceof Date) {
      out = 'Date(' + v.getTime() + ')';
    } else if (v instanceof Map) {
      out = 'Map(' + [...v].map(([k, x]) => walk(k) + ':' + walk(x)).sort().join(',') + ')';
    } else if (v instanceof Set) {
      out = 'Set(' + [...v].map(walk).sort().join(',') + ')';
    } else {
      out = '{' + Object.keys(v).sort()
        .map((k) => JSON.stringify(k) + ':' + walk(v[k])).join(',') + '}';
    }

    seen.delete(v);
    return out;
  })(value);
}

Step 8

Complexity, in sentences

TimeSpace
Cache hitO(key)—
Cache missO(key) + the functionO(key + result)
Whole cache—O(distinct calls) — unbounded by default

O(key) is the cost of building the key, and it is the number people forget. A single-argument primitive key is O(1); JSON.stringify of a large object is O(size of the object), on every call including hits. Memoizing a cheap function with an expensive key makes it slower.

The space row is the one that causes incidents. A memoized function keyed on user input has an unbounded cache, holds every result forever, and — with a Map — holds every key object forever too. Cap it, or use a WeakMap, or accept that it is a leak.

Say this in the interview

“A closure over a Map, keyed on the arguments. I use cache.has rather than a truthiness check, so a function returning undefined is still cached, and a Map rather than an object so numeric and string keys do not collide and nothing is inherited from the prototype. The real question is the key: one primitive argument can be the key directly; several want a nested Map or a delimiter join; an object wants a WeakMap if identity is what matters, or a stable sorted stringify if contents are. And I would cap the size, because otherwise it is an unbounded cache keyed on whatever the user types.”

    if (cache.has(k)) return cache.get(k);
    const r = fn(...args);
    cache.set(k, r);
    return r;
  };
}

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

// (a) it reads the clock
const now = memoize(() => Date.now());
const first = now();
await sleep(30);
console.log('a) time is frozen: same value 30 ms later?', now() === first);

// (b) it reads a mutable object
const config = { mode: 'dark' };
const describe = memoize((label) => label + ' in ' + config.mode);
console.log('');
console.log('b)', describe('theme'));
config.mode = 'light';
console.log('  ', describe('theme'), '<- stale, config changed');

// (c) it returns a mutable object -- callers share it
const build = memoize((n) => ({ items: [n] }));
const one = build(1);
const two = build(1);
one.items.push('MUTATED BY ANOTHER CALLER');
console.log('');
console.log('c) two callers, same object?', one === two);
console.log('  ', JSON.stringify(two));

console.log('');
console.log('Memoization is only valid for a pure function. If the');
console.log('result depends on anything outside the arguments, that');
console.log('thing has to be part of the KEY -- or you must not');
console.log('memoize at all.');
console.log('');
console.log('Fix for (b): key on the config too.');
const fixed = memoize((label, mode) => label + ' in ' + mode);
console.log('  ', fixed('theme', 'dark'), '/', fixed('theme', 'light'));
console.log('Fix for (c): return a copy, or freeze the result.');

2. Recursing into the unmemoized function

Broken — the memo does nothing
function memoize(fn) {
  const cache = new Map();
  return (...args) => {
    const k = String(args);
    if (cache.has(k)) return cache.get(k);
    const r = fn(...args);
    cache.set(k, r);
    return r;
  };
}

// the recursive calls go to `fib`, not to the memoized wrapper
let brokenCalls = 0;
function fib(n) {
  brokenCalls++;
  return n < 2 ? n : fib(n - 1) + fib(n - 2);
}
const memoFib = memoize(fib);

memoFib(28);
console.log('memoize(fib)(28) -- inner calls:',
            brokenCalls.toLocaleString('en-US'));
console.log('  the wrapper cached exactly ONE entry: the top-level call.');
console.log('  Every recursive call went straight to the raw function.');

// the fix: make the function call the wrapper
let fixedCalls = 0;
const goodFib = memoize((n) => {
  fixedCalls++;
  return n < 2 ? n : goodFib(n - 1) + goodFib(n - 2);
});
goodFib(28);
console.log('');
console.log('self-referential version   -- inner calls:', fixedCalls);

// or pass the memoized version in
let injectedCalls = 0;
function memoizeRecursive(fn) {
  const cache = new Map();
  const wrapped = (...args) => {
    const k = String(args);
    if (cache.has(k)) return cache.get(k);
    const r = fn(wrapped, ...args);      // hand the wrapper to the function
    cache.set(k, r);
    return r;
  };
  return wrapped;
}
const injected = memoizeRecursive((self, n) => {
  injectedCalls++;
  return n < 2 ? n : self(n - 1) + self(n - 2);
});
injected(28);
console.log('injection version          -- inner calls:', injectedCalls);
console.log('');
console.log('The second is cleaner: the function does not have to know');
console.log('the name of its own memoized wrapper.');

3. An unbounded cache on user input

Broken — a leak with a friendly name
function memoize(fn) {
  const cache = new Map();
  const w = (...a) => {
    const k = String(a);
    if (cache.has(k)) return cache.get(k);
    const r = fn(...a);
    cache.set(k, r);
    return r;
  };
  w.cache = cache;
  return w;
}

// a search box: every keystroke is a distinct key
const search = memoize((q) => ({ query: q, results: new Array(50).fill(q) }));

const typed = 'javascript interview questions';
for (let i = 1; i <= typed.length; i++) search(typed.slice(0, i));

console.log('one user typed one phrase.');
console.log('  cache entries:', search.cache.size);
console.log('  hit rate     : 0% -- every prefix is unique');

console.log('');
console.log('Now multiply by every user, every session, forever. The');
console.log('cache only grows, and every entry holds a 50-item array.');

// bounded version
function memoizeLRU(fn, max) {
  const cache = new Map();
  const w = (...a) => {
    const k = String(a);
    if (cache.has(k)) { const v = cache.get(k); cache.delete(k); cache.set(k, v); return v; }
    const r = fn(...a);
    cache.set(k, r);
    if (cache.size > max) cache.delete(cache.keys().next().value);
    return r;
  };
  w.cache = cache;
  return w;
}
const bounded = memoizeLRU((q) => ({ query: q }), 10);
for (let i = 1; i <= typed.length; i++) bounded(typed.slice(0, i));
console.log('');
console.log('bounded to 10 entries ->', bounded.cache.size, 'entries held');

console.log('');
console.log('Before memoizing, ask: how many DISTINCT inputs will this');
console.log('ever see? If the answer is "as many as the user types",');
console.log('you need a bound, a TTL, or no cache at all.');

4. Sharing one cache across instances

Broken — one instance served another instance's answer
function memoize(fn) {
  const cache = new Map();
  return function (...args) {
    const k = JSON.stringify(args);
    if (cache.has(k)) return cache.get(k);
    const r = fn.apply(this, args);
    cache.set(k, r);
    return r;
  };
}

// memoizing on the PROTOTYPE means one cache for every instance
class Account {
  constructor(owner, balance) { this.owner = owner; this.balance = balance; }
}
Account.prototype.report = memoize(function (label) {
  return label + ': ' + this.owner + ' has ' + this.balance;
});

const ada = new Account('Ada', 100);
const grace = new Account('Grace', 500);

console.log('ada  :', ada.report('balance'));
console.log('grace:', grace.report('balance'), '  <- Ada has leaked in');
console.log('');
console.log('The key was JSON of the ARGUMENTS -- "balance" in both');
console.log('cases. `this` is not part of the key, so the second');
console.log('instance got the cached answer from the first.');

// fix A: a per-instance cache, created in the constructor
class FixedA {
  constructor(owner, balance) {
    this.owner = owner;
    this.balance = balance;
    this.report = memoize((label) => label + ': ' + this.owner + ' has ' + this.balance);
  }
}
console.log('');
console.log('fix A -- per-instance memo:');
console.log(' ', new FixedA('Ada', 100).report('balance'));
console.log(' ', new FixedA('Grace', 500).report('balance'));

// fix B: put `this` in the key with a WeakMap of caches
function memoizePerReceiver(fn) {
  const caches = new WeakMap();
  return function (...args) {
    if (!caches.has(this)) caches.set(this, new Map());
    const cache = caches.get(this);
    const k = JSON.stringify(args);
    if (cache.has(k)) return cache.get(k);
    const r = fn.apply(this, args);
    cache.set(k, r);
    return r;
  };
}
class FixedB {
  constructor(owner, balance) { this.owner = owner; this.balance = balance; }
}
FixedB.prototype.report = memoizePerReceiver(function (label) {
  return label + ': ' + this.owner + ' has ' + this.balance;
});
console.log('');
console.log('fix B -- one cache per receiver, shared method:');
console.log(' ', new FixedB('Ada', 100).report('balance'));
console.log(' ', new FixedB('Grace', 500).report('balance'));

Step 10

Practice before you move on

  1. Write the six-line version from memory, then immediately say two things that are wrong with JSON.stringify as the key.
  2. Write the single-argument version and explain why Map beats a plain object — numeric keys, inherited keys, and has versus undefined.
  3. Write the LRU version. Step 6. The delete-then-set move is worth having in your fingers; it comes up in cache questions of every kind.
  4. Write the async version and explain why it caches the promise rather than the value. Then explain the .catch that deletes the entry.
  5. Write memoizeRecursive — Step 9's injection version — and use it to memoize a recursive grid-paths or edit-distance function. That is the bridge from memoization to dynamic programming.
What to carry forward

The cache is trivial; the key is the design. Every follow-up question on this topic — multiple arguments, objects, promises, invalidation, leaks — is answered by deciding what identifies a call and how long that answer stays true. Ask those two questions before you write a line.
