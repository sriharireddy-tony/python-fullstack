Object & data shaping · Problem 07

Paginate, Sort and Filter an Array of Records

The endpoint you have written a dozen times. Three things make it a real question: the operations must happen in the right order, sort mutates and stringifies, and offset pagination silently skips rows when the data changes underneath it.

Step 1

The problem

Version 0 — filter, sort, slice
const users = [
  { id: 1, name: 'Ada', age: 36, city: 'London', active: true },
  { id: 2, name: 'Grace', age: 45, city: 'New York', active: true },
  { id: 3, name: 'Alan', age: 41, city: 'London', active: false },
  { id: 4, name: 'Edsger', age: 42, city: 'Amsterdam', active: true },
  { id: 5, name: 'Barbara', age: 38, city: 'London', active: true },

function query(records, { filter = {}, sortBy, sortDir = 'asc', page = 1, perPage = 10 }) {
  // 1. FILTER first -- so the total and the pages are of the filtered set
  let rows = records.filter((r) =>
    Object.entries(filter).every(([k, v]) => r[k] === v));

  const total = rows.length;

  // 2. SORT the filtered rows
  if (sortBy) {
    rows = [...rows].sort((a, b) => {
      const x = a[sortBy], y = b[sortBy];
      const cmp = x < y ? -1 : x > y ? 1 : 0;
      return sortDir === 'desc' ? -cmp : cmp;
    });
  }

  // 3. SLICE the page out of the sorted rows
  const start = (page - 1) * perPage;
  const data = rows.slice(start, start + perPage);

  return {
    data,
    total,
    page,
    perPage,
    totalPages: Math.max(1, Math.ceil(total / perPage)),
    hasNext: start + perPage < total,
  };
}

The order is not negotiable

Filter, then sort, then slice.

Filter first because the total and the page count must describe the filtered set, and because there is no point sorting rows you are about to throw away.

Sort before slicing because slicing an unsorted array gives you an arbitrary page. Getting those two backwards is the single most common bug in this code, and it is shown in Step 2.

Step 4

The better idea — and the sort traps

Three things about Array.prototype.sort

It mutates. rows.sort(...) reorders the caller's array. Always [...rows].sort(...) or rows.toSorted(...).

Without a comparator it stringifies. [10, 9, 1].sort() gives [1, 10, 9].

It is stable since ES2019. Equal elements keep their relative order, which is what makes multi-column sorting work by sorting repeatedly — though doing it in one comparator is better.

Version 1 — the comparator, done properly

multi-column sorting in one comparator
// A comparator factory: a list of { key, dir, type } specs.
function makeComparator(specs) {
  const compare = {
    number: (a, b) => a - b,
    string: (a, b) => String(a).localeCompare(String(b)),
    date: (a, b) => new Date(a) - new Date(b),
    boolean: (a, b) => (a === b ? 0 : a ? 1 : -1),
  };

  return (rowA, rowB) => {
    for (const { key, dir = 'asc', type = 'string' } of specs) {
      const a = rowA[key], b = rowB[key];

      // nulls always last, whichever direction
      const aNull = a === null || a === undefined || a === '';
      const bNull = b === null || b === undefined || b === '';
      if (aNull && bNull) continue;
      if (aNull) return 1;
      if (bNull) return -1;

      const cmp = compare[type](a, b);
      if (cmp !== 0) return dir === 'desc' ? -cmp : cmp;
    }
    return 0;
  };
}

function makeComparator(specs) {
  const cmp = {
    number: (a, b) => a - b,
    string: (a, b) => String(a).localeCompare(String(b)),
    date: (a, b) => new Date(a) - new Date(b),
    boolean: (a, b) => (a === b ? 0 : a ? 1 : -1),
  };
  return (ra, rb) => {
    for (const { key, dir = 'asc', type = 'string' } of specs) {
      const a = ra[key], b = rb[key];
      const an = a === null || a === undefined || a === '';
      const bn = b === null || b === undefined || b === '';
      if (an && bn) continue;
      if (an) return 1;
      if (bn) return -1;
      const c = cmp[type](a, b);
      if (c !== 0) return dir === 'desc' ? -c : c;
    }
    return 0;
  };
}

const OPERATORS = {
  eq: (v, t) => v === t,

  gt: (v, t) => v > t,

function query(records, options = {}) {
  const {

    maxPerPage = 100,

  const size = Math.min(Math.max(1, Math.floor(perPage) || 1), maxPerPage);

  // 1. FILTER
  let rows = records;
  for (const { key, op = 'eq', value } of where) {
    const test = OPERATORS[op];
    if (!test) throw new Error('unknown operator: ' + op);
    rows = rows.filter((r) => test(r[key], value));
  }

  if (search && searchIn.length) {
    const needle = search.toLowerCase();
    rows = rows.filter((r) => searchIn.some((k) =>
      String(r[k] ?? '').toLowerCase().includes(needle)));
  }

  const total = rows.length;
  const totalPages = Math.max(1, Math.ceil(total / size));
  const current = Math.min(Math.max(1, Math.floor(page) || 1), totalPages);

  // 2. SORT -- on a copy, always with a unique tie-break
  if (sort.length) {
    const specs = sort.some((s) => s.key === 'id')
      ? sort
      : [...sort, { key: 'id', type: 'number' }];
    rows = [...rows].sort(makeComparator(specs));
  }

  // 3. SLICE
  const from = (current - 1) * size;
  const data = rows.slice(from, from + size);

  return {
    data,
    meta: {
      total, page: current, perPage: size, totalPages,
      from: total === 0 ? 0 : from + 1,
      to: from + data.length,
      hasPrev: current > 1,
      hasNext: current < totalPages,
    },
  };
}

// CURSOR pagination: "give me rows after this one"
function cursorPage(list, { after = null, size = 3 }) {
  const sorted = [...list].sort((a, b) => a.id - b.id);
  const start = after === null ? 0 : sorted.findIndex((r) => r.id === after) + 1;
  const data = sorted.slice(start, start + size);
  return {
    data,
    nextCursor: data.length === size ? data[data.length - 1].id : null,
  };
}

const byScoreDesc = (a, b) => b.score - a.score || a.id - b.id;

// WRONG: the cursor is only the score
function badCursor(list, { afterScore = null, size = 2 }) {
  const sorted = [...list].sort(byScoreDesc);
  const data = (afterScore === null
    ? sorted
    : sorted.filter((r) => r.score < afterScore)
  ).slice(0, size);
  return { data, next: data.length ? data[data.length - 1].score : null };
}

// RIGHT: the cursor is (score, id) -- the full sort key
function goodCursor(list, { after = null, size = 2 }) {
  const sorted = [...list].sort(byScoreDesc);
  const data = (after === null
    ? sorted
    : sorted.filter((r) =>
        r.score < after.score || (r.score === after.score && r.id > after.id))
  ).slice(0, size);
  const last = data[data.length - 1];
  return { data, next: data.length === size ? { score: last.score, id: last.id } : null };
}

Step 8

Complexity, in sentences

StepCost
filterO(n) per predicate
sortO(m log m), m = rows that survived the filter
sliceO(perPage)
totalO(n · p + m log m)

Filtering first is what makes m small, and m log m is the dominant term — so the order is a complexity decision as well as a correctness one.

The real point about complexity here, though, is different: all of this belongs in the database. Doing it in JavaScript means loading every row into memory to return twenty of them. It is the right answer for an in-memory table, a client-side data grid, or a fixture — and the wrong answer for a table with a million rows, where WHERE, ORDER BY and LIMIT use indexes you cannot replicate in an array.

Say this in the interview

“Filter, then sort, then slice — in that order, because the total has to describe the filtered set and slicing before sorting gives you an arbitrary page. I sort a copy, never the input, always with an explicit comparator, and always with a unique tie-break, because a stable sort only preserves the order it was given — if the input order can change, two equal rows can swap between requests and a row is missed or duplicated. I clamp the page and page size. And I would say that offset pagination is inherently racy with concurrent writes: a delete shifts everything up and a row is skipped, so for an append-heavy feed I would use a cursor carrying the full sort key. In production, all of this goes in the query, not in JavaScript.”

// memoise the filter step, since the sort key did not change
let lastSearch = null, lastFiltered = null;
function queryMemo(records, search, sortBy) {
  const needle = search.toLowerCase();
  if (needle !== lastSearch) {
    lastSearch = needle;
    lastFiltered = needle
      ? records.filter((r) => r.name.toLowerCase().includes(needle))
      : records;
  }
  const sorted = [...lastFiltered].sort((a, b) => a[sortBy] - b[sortBy] || a.id - b.id);
  return { data: sorted.slice(0, 20), total: lastFiltered.length };
}

Practice before you move on

  2. Write the comparator factory. Step 6 — multi-column, typed, nulls last, with a unique tie-break.

What to carry forward

Order of operations, and a total order. Filter before sort before slice; and the sort must be a total order, meaning a unique tie-break, or pagination is undefined even with a stable sort. Both bugs are invisible in a test fixture and both show up as “sometimes a row is missing” in production, which is the hardest kind of bug report to act on.
